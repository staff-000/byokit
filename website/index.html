import * as byokit from
  '@byokit/core/catalog';

const [model] = byokit.getModels();
const client = byokit.openAI({
  model, apiKey: () => key,
});
const completion = await
  client.chat.completions.create({
    messages: [{
      role: byokit.Role.User, content: 'Hello',
    }],
  });
